<script>

    const STORAGE_KEY = 'shiftScheduleData';

    const MONTHS = ['Январь','Февраль','Март','Апрель','Май','Июнь',

                    'Июль','Август','Сентябрь','Октябрь','Ноябрь','Декабрь'];

 

    let data = { weeks: [] };

    let weekCounter = 0;

    let dayCounter = 0;

 

    function loadData() {

        try {

            const saved = localStorage.getItem(STORAGE_KEY);

            if (saved) {

                data = JSON.parse(saved);

                data.weeks.forEach(w => {

                    if (typeof w.closed === 'undefined') w.closed = false;

                    weekCounter = Math.max(weekCounter, w.id + 1);

                    w.days.forEach(d => {

                        if (typeof d.paid === 'undefined') d.paid = null;

                        dayCounter = Math.max(dayCounter, d.id + 1);

                    });

                });

            }

        } catch (e) { console.error('Ошибка загрузки:', e); }

    }

 

    function saveData() {

        try { localStorage.setItem(STORAGE_KEY, JSON.stringify(data)); }

        catch (e) { console.error('Ошибка сохранения:', e); }

    }

 

    function getCurrentMonth() { return MONTHS[new Date().getMonth()]; }

 

    function createWeek() {

        data.weeks.push({ id: weekCounter++, closed: false, days: [] });

        saveData();

        render();

    }

 

    function deleteWeek(weekId) {

        if (!confirm('Удалить неделю со всеми днями?')) return;

        data.weeks = data.weeks.filter(w => w.id !== weekId);

        saveData();

        render();

    }

 

    function toggleFinishWeek(weekId) {

        const week = data.weeks.find(w => w.id === weekId);

        if (!week) return;

        if (!week.closed) {

            if (!confirm('Завершить неделю и рассчитать итог?')) return;

            week.closed = true;

        } else {

            if (!confirm('Возобновить неделю для редактирования?')) return;

            week.closed = false;

        }

        saveData();

        render();

    }

 

    function createDay(weekId) {

        const week = data.weeks.find(w => w.id === weekId);

        if (!week) return;

        if (week.closed) { alert('Неделя завершена. Возобновите её.'); return; }

        week.days.push({

            id: dayCounter++,

            date: '', night: 0, day: 0, evening: 0,

            notes: '', paid: null

        });

        saveData();

        render();

    }

 

    function deleteDay(weekId, dayId) {

        if (!confirm('Удалить этот день?')) return;

        const week = data.weeks.find(w => w.id === weekId);

        if (!week) return;

        week.days = week.days.filter(d => d.id !== dayId);

        saveData();

        render();

    }

 

    function updateDay(weekId, dayId, field, value) {

        const week = data.weeks.find(w => w.id === weekId);

        if (!week) return;

        const day = week.days.find(d => d.id === dayId);

        if (!day) return;

 

        if (field === 'night' || field === 'day' || field === 'evening') {

            let num = parseInt(value, 10);

            if (isNaN(num) || num < 0) num = 0;

            day[field] = num;

        } else {

            day[field] = value;

        }

        saveData();

    }

 

    function openPaidDialog(weekId, dayId) {

        const week = data.weeks.find(w => w.id === weekId);

        if (!week) return;

        const day = week.days.find(d => d.id === dayId);

        if (!day) return;

 

        const current = day.paid == null ? '' : day.paid;

        const input = prompt('Введите сумму, которую вам оплатили за смену (руб.):', current);

        if (input === null) return;

 

        const cleaned = input.replace(/[^0-9]/g, '');

        day.paid = cleaned === '' ? null : parseInt(cleaned, 10);

        saveData();

        render();

    }

 

    function getDayTotals(day) {

        const carsTotal = (day.night || 0) + (day.day || 0) + (day.evening || 0);

        return { carsTotal, moneyTotal: carsTotal * 300 };

    }

 

    function getPaidStatus(day) {

        if (day.paid == null) return null;

        const { moneyTotal } = getDayTotals(day);

        if (day.paid === moneyTotal) return 'match';

        if (day.paid < moneyTotal) return 'less';

        return 'more';

    }

 

    function getWeekTotals(week) {

        let cars = 0, money = 0, paid = 0, paidCount = 0;

        week.days.forEach(d => {

            const t = getDayTotals(d);

            cars += t.carsTotal;

            money += t.moneyTotal;

            if (d.paid != null) { paid += d.paid; paidCount++; }

        });

        return { cars, money, paid, paidCount, diff: paid - money };

    }

 

    function render() {

        const container = document.getElementById('weeksContainer');

        container.innerHTML = '';

 

        if (data.weeks.length === 0) {

            container.innerHTML = '<div class="empty-text">Пока нет ни одной недели. Нажмите «Создать неделю».</div>';

        }

 

        data.weeks.forEach((week, wIndex) => {

            const weekEl = document.createElement('div');

            weekEl.className = 'week-block' + (week.closed ? ' closed' : '');

            const month = getCurrentMonth();

 

            weekEl.innerHTML = `

                <div class="week-header">

                    <div class="week-title">📆 ${month} — Неделя ${wIndex + 1}${week.closed ? ' ✅' : ''}</div>

                    <div class="week-actions">

                        <button class="btn btn-small btn-finish ${week.closed ? 'finished' : ''}"

                                onclick="toggleFinishWeek(${week.id})">

                            ${week.closed ? '↩ Возобновить' : '🏁 Завершить неделю'}

                        </button>

                        <button class="btn btn-small btn-danger" onclick="deleteWeek(${week.id})">🗑</button>

                    </div>

                </div>

                <div class="days-grid" id="days-${week.id}"></div>

                <div id="total-${week.id}"></div>

            `;

            container.appendChild(weekEl);

 

            const daysGrid = weekEl.querySelector(`#days-${week.id}`);

 

            week.days.forEach(day => {

                daysGrid.appendChild(buildDayCard(week.id, day, week.closed));

            });

 

            if (!week.closed) {

                const addTile = document.createElement('div');

                addTile.className = 'add-day-tile';

                addTile.textContent = '＋';

                addTile.title = 'Добавить день';

                addTile.onclick = () => createDay(week.id);

                daysGrid.appendChild(addTile);

            }

 

            if (week.closed) {

                const totalsEl = weekEl.querySelector(`#total-${week.id}`);

                totalsEl.appendChild(buildWeekTotal(week));

            }

        });

    }

 

    function buildWeekTotal(week) {

        const t = getWeekTotals(week);

        const el = document.createElement('div');

        el.className = 'week-total show';

 

        let diffText = '—';

        let diffColor = 'rgba(255,255,255,0.85)';

        if (t.paidCount > 0) {

            const sign = t.diff > 0 ? '+' : '';

            diffText = sign + t.diff + ' руб.';

            if (t.diff === 0) diffColor = '#b6ffd0';

            else if (t.diff > 0) diffColor = '#d5b6ff';

            else diffColor = '#ffb6b6';

        }

 

        el.innerHTML = `

            <div class="week-total-title">🏁 Итог за неделю</div>

            <div class="week-total-grid">

                <span class="k">Всего машин:</span>

                <span class="v">${t.cars}</span>

                <span class="k">Расчётная сумма:</span>

                <span class="v highlight">${t.money} руб.</span>

                <span class="k">Оплачено:</span>

                <span class="v">${t.paidCount > 0 ? t.paid + ' руб.' : '—'}</span>

                <span class="k">Разница:</span>

                <span class="v" style="color:${diffColor}">${diffText}</span>

            </div>

        `;

        return el;

    }

 

    function buildDayCard(weekId, day, weekClosed) {

        const card = document.createElement('div');

        card.className = 'day-card';

 

        const nightMoney = day.night * 300;

        const dayMoney = day.day * 300;

        const eveningMoney = day.evening * 300;

        const { carsTotal } = getDayTotals(day);

        const status = getPaidStatus(day);

 

        let paidBtnClass = 'paid-btn';

        let paidBtnText = '💰 Оплачено';

        if (status === 'match') { paidBtnClass += ' status-match'; paidBtnText = `✅ ${day.paid} руб.`; }

        else if (status === 'less') { paidBtnClass += ' status-less'; paidBtnText = `⬇ ${day.paid} руб.`; }

        else if (status === 'more') { paidBtnClass += ' status-more'; paidBtnText = `⬆ ${day.paid} руб.`; }

 

        const disabled = weekClosed ? 'disabled' : '';

 

        card.innerHTML = `

            <div class="day-top">

                <input type="text" class="day-number-input" placeholder="28.09"

                       value="${escapeHtml(day.date)}" ${disabled}

                       oninput="updateDay(${weekId}, ${day.id}, 'date', this.value)">

                <div class="day-top-btns">

                    <button class="icon-btn notes-indicator" title="Примечания"

                            onclick="toggleNotes(this)">📝</button>

                    ${!weekClosed ? `<button class="icon-btn" title="Удалить день"

                            onclick="deleteDay(${weekId}, ${day.id})">✕</button>` : ''}

                </div>

            </div>

            <div class="shift-row">

                <span class="shift-label">Ночь</span>

                <input type="number" class="cars-input" min="0" placeholder="0"

                       value="${day.night || ''}" ${disabled}

                       oninput="onCarsInput(this, ${weekId}, ${day.id}, 'night')">

                <span class="cars-word">машин</span>

                <span class="money">${nightMoney} руб.</span>

            </div>

            <div class="shift-row">

                <span class="shift-label">День</span>

                <input type="number" class="cars-input" min="0" placeholder="0"

                       value="${day.day || ''}" ${disabled}

                       oninput="onCarsInput(this, ${weekId}, ${day.id}, 'day')">

                <span class="cars-word">машин</span>

                <span class="money">${dayMoney} руб.</span>

            </div>

            <div class="shift-row">

                <span class="shift-label">Вечер</span>

                <input type="number" class="cars-input" min="0" placeholder="0"

                       value="${day.evening || ''}" ${disabled}

                       oninput="onCarsInput(this, ${weekId}, ${day.id}, 'evening')">

                <span class="cars-word">машин</span>

                <span class="money">${eveningMoney} руб.</span>

                    ${paidBtnText}

                </button>

            </div>

                <textarea class="notes-textarea" placeholder="Дополнительные комментарии..."

                          ${disabled}

                          oninput="updateDay(${weekId}, ${day.id}, 'notes', this.value)">${escapeHtml(day.notes)}</textarea>

            </div>

        `;

        return card;

    }

 

    function onCarsInput(input, weekId, dayId, field) {

        let val = input.value.replace(/[^0-9]/g, '');

        input.value = val;

        updateDay(weekId, dayId, field, val);

        recalcCard(input.closest('.day-card'), weekId, dayId);

    }

 

    function recalcCard(card, weekId, dayId) {

        const week = data.weeks.find(w => w.id === weekId);

        if (!week) return;

        const day = week.days.find(d => d.id === dayId);

        if (!day) return;

 

        const rows = card.querySelectorAll('.shift-row');

        const values = [day.night, day.day, day.evening];

 

        rows.forEach((row, i) => {

            row.querySelector('.money').textContent = (values[i] * 300) + ' руб.';

        });

 

        const { carsTotal } = getDayTotals(day);

        card.querySelector('.total-value').textContent = carsTotal;

 

        const paidBtn = card.querySelector('.paid-btn');

        if (paidBtn && day.paid != null) {

            paidBtn.classList.remove('status-match', 'status-less', 'status-more');

            const status = getPaidStatus(day);

            if (status === 'match') {

                paidBtn.classList.add('status-match');

                paidBtn.textContent = `✅ ${day.paid} руб.`;

            } else if (status === 'less') {

                paidBtn.classList.add('status-less');

                paidBtn.textContent = `⬇ ${day.paid} руб.`;

            } else if (status === 'more') {

                paidBtn.classList.add('status-more');

                paidBtn.textContent = `⬆ ${day.paid} руб.`;

            }

        }

    }

 

    function toggleNotes(btn) {

        const card = btn.closest('.day-card');

        card.querySelector('.notes-block').classList.toggle('open');

    }

 

    function escapeHtml(str) {

        if (!str) return '';

        return String(str)

            .replace(/&/g, '&amp;').replace(/</g, '&lt;')

            .replace(/>/g, '&gt;').replace(/"/g, '&quot;')

            .replace(/'/g, '&#39;');

    }
